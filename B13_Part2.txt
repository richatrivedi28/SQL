-- Window Functions-- 

select rank() over(order by imdb_rating desc) as rating, title,industry, imdb_rating from movies;

select dense_rank() over(order by imdb_rating desc) as rating, title,industry, imdb_rating from movies;

select dense_rank() over(partition by industry order by imdb_rating desc)  as rating, title,industry, imdb_rating from movies;


create table temp_table as
select dense_rank() over(order by imdb_rating desc) as rating, title,industry, imdb_rating from movies;

select * from temp_table where rating =2;

With categorized as (
 select dense_rank() over(order by imdb_rating desc) as rating, title,industry, imdb_rating from movies
 )
 select imdb_rating, rating from categorized where rating =2 ;
 
 
 Select * from financials;
 
 create table new_table as 
 select *, 
 case when unit = "Thousands" then revenue/1000
	  when unit = "Billions" then revenue*1000
      else revenue
end as revenue_mn
from financials;

select * from new_table;

select *,
case when revenue_mn < 5000 then "Low"
	 when revenue_mn < 10000 then "Medium"
     else "High"
end as revenue_category
from new_table;

with categorized as (
select *,
case when revenue_mn < 5000 then "Low"
	 when revenue_mn < 10000 then "Medium"
     else "High"
end as revenue_category
from new_table)
select * from categorized where revenue_category = "High";

-- find the total quantity of each product sold and display only those where total quantity > 500

select * from orderdetails;
 
select productcode, sum(quantityordered) as sum_qty 
from orderdetails 
group by productcode
having sum_qty > 500
order by sum_qty desc;
